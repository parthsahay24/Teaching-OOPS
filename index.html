CONCEPT 01

Classes & Objects

A class is a blueprint that defines attributes (data) and methods (behavior). An object is a concrete instance of that class living in memory.

__init__ / constructor attributes methods instantiation self / this
PYTHON 
# Define a class
class Car:
    # Constructor
    def __init__(self, brand, speed):
        self.brand = brand     # attribute
        self.speed = speed

    def describe(self):
        print(f"Brand: {self.brand}, Speed: {self.speed} km/h")

    def accelerate(self, amount):
        self.speed += amount
        print(f"Now going {self.speed} km/h!")


# Create objects (instances)
car1 = Car("Tesla", 120)
car2 = Car("BMW", 150)

car1.describe()
car1.accelerate(30)
car2.describe()
C++ 
// Define a class
#include <iostream>
#include <string>
using namespace std;

class Car {
public:
    string brand;
    int speed;

    // Constructor
    Car(string b, int s) : brand(b), speed(s) {}

    void describe() {
        cout << "Brand: " << brand
             << ", Speed: " << speed << " km/h\n";
    }

    void accelerate(int amount) {
        speed += amount;
        cout << "Now going " << speed << " km/h!\n";
    }
};

int main() {
    Car car1("Tesla", 120);
    Car car2("BMW", 150);
    car1.describe();
    car1.accelerate(30);
    car2.describe();
}

CONCEPT 02

Inheritance

A child class inherits attributes and methods from a parent class, enabling code reuse and hierarchical classification. The child can also override or extend parent behavior.

parent → child super() method override is-a relationship code reuse
PYTHON 
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        print(f"{self.name} makes a sound")

    def breathe(self):
        print(f"{self.name} breathes oxygen")


# Dog inherits from Animal
class Dog(Animal):
    def __init__(self, name, breed):
        super().__init__(name)   # call parent __init__
        self.breed = breed

    # Override speak()
    def speak(self):
        print(f"{self.name} says: Woof!")


class Cat(Animal):
    def speak(self):
        print(f"{self.name} says: Meow!")


dog = Dog("Bruno", "Labrador")
cat = Cat("Whiskers")

dog.speak()
dog.breathe()   # inherited method
cat.speak()
C++ 
class Animal {
public:
    string name;
    Animal(string n) : name(n) {}

    virtual void speak() {
        cout << name << " makes a sound\n";
    }
    void breathe() {
        cout << name << " breathes oxygen\n";
    }
};

// Dog inherits from Animal
class Dog : public Animal {
public:
    string breed;
    Dog(string n, string b)
        : Animal(n), breed(b) {}   // call parent ctor

    void speak() override {
        cout << name << " says: Woof!\n";
    }
};

class Cat : public Animal {
public:
    Cat(string n) : Animal(n) {}
    void speak() override {
        cout << name << " says: Meow!\n";
    }
};

int main() {
    Dog dog("Bruno", "Labrador");
    Cat cat("Whiskers");
    dog.speak();
    dog.breathe();   // inherited
    cat.speak();
}
OUTPUT

Bruno breathes oxygen
Whiskers says: Meow!

CONCEPT 03

Polymorphism

"Many forms" — the same method name behaves differently across different classes. Achieved via method overriding (runtime) and method overloading (compile-time in C++).

one interface, many forms runtime dispatch virtual methods duck typing (Python) overloading (C++)
PYTHON 
class Shape:
    def area(self):
        return 0

class Circle(Shape):
    def __init__(self, r):
        self.r = r
    def area(self):
        return 3.14 * self.r ** 2

class Rectangle(Shape):
    def __init__(self, w, h):
        self.w = w; self.h = h
    def area(self):
        return self.w * self.h

class Triangle(Shape):
    def __init__(self, b, h):
        self.b = b; self.h = h
    def area(self):
        return 0.5 * self.b * self.h


# Polymorphic loop — same call, different result
shapes = [Circle(5), Rectangle(4,6), Triangle(3,8)]
for s in shapes:
    print(f"{type(s).__name__} area: {s.area()}")
C++ 
class Shape {
public:
    virtual double area() { return 0; }
    virtual ~Shape() {}
};

class Circle : public Shape {
    double r;
public:
    Circle(double r) : r(r) {}
    double area() override { return 3.14 * r * r; }
};

class Rectangle : public Shape {
    double w, h;
public:
    Rectangle(double w, double h) : w(w), h(h) {}
    double area() override { return w * h; }
};

class Triangle : public Shape {
    double b, h;
public:
    Triangle(double b, double h) : b(b), h(h) {}
    double area() override { return 0.5 * b * h; }
};

int main() {
    Shape* shapes[] = {
        new Circle(5),
        new Rectangle(4,6),
        new Triangle(3,8)
    };
    for (auto s : shapes)
        cout << "Area: " << s->area() << "\n";
}
OUTPUT
Circle area: 78.5
Rectangle area: 24
Triangle area: 12.0

CONCEPT 04

Encapsulation

Bundling data and methods together while restricting direct access to internal state. Achieved through access modifiers and getter/setter methods, protecting data integrity.

data hiding private / public getters & setters _ prefix (Python) @property
PYTHON 
class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner
        self.__balance = balance  # private

    # Getter (read-only access)
    @property
    def balance(self):
        return self.__balance

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            print(f"Deposited ₹{amount}")
        else:
            print("Invalid amount!")

    def withdraw(self, amount):
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            print(f"Withdrawn ₹{amount}")
        else:
            print("Insufficient funds!")


acc = BankAccount("Arjun", 5000)
acc.deposit(2000)
acc.withdraw(1000)
print(f"Balance: ₹{acc.balance}")
# acc.__balance → AttributeError ✗
C++ 
class BankAccount {
private:          // hidden from outside
    string owner;
    double balance;

public:
    BankAccount(string o, double b)
        : owner(o), balance(b) {}

    // Getter
    double getBalance() { return balance; }

    void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            cout << "Deposited ₹" << amount << "\n";
        } else
            cout << "Invalid amount!\n";
    }

    void withdraw(double amount) {
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            cout << "Withdrawn ₹" << amount << "\n";
        } else
            cout << "Insufficient funds!\n";
    }
};

int main() {
    BankAccount acc("Arjun", 5000);
    acc.deposit(2000);
    acc.withdraw(1000);
    cout << "Balance: ₹" << acc.getBalance() << "\n";
    // acc.balance → compile error ✗
}

Balance: ₹6000

CONCEPT 05

Abstraction

Hiding implementation complexity and exposing only essential features. Abstract classes define a contract (interface) that all subclasses must fulfill, without dictating how.

abstract class ABC / pure virtual interface contract hide complexity @abstractmethod
PYTHON 
from abc import ABC, abstractmethod

class Vehicle(ABC):  # abstract base class

    @abstractmethod
    def start(self):
        pass          # must be implemented

    @abstractmethod
    def fuel_type(self):
        pass

    def description(self):  # concrete method
        print(f"Vehicle runs on: {self.fuel_type()}")


class ElectricCar(Vehicle):
    def start(self):
        print("Electric car: silent start ⚡")
    def fuel_type(self):
        return "Electricity"


class PetrolCar(Vehicle):
    def start(self):
        print("Petrol car: Vroom! 🔥")
    def fuel_type(self):
        return "Petrol"


# v = Vehicle()  → TypeError: can't instantiate
ev = ElectricCar()
pc = PetrolCar()

ev.start(); ev.description()
pc.start(); pc.description()
C++ 
#include <iostream>
using namespace std;

// Abstract base class
class Vehicle {
public:
    // Pure virtual = must override
    virtual void start() = 0;
    virtual string fuelType() = 0;

    void description() {   // concrete method
        cout << "Runs on: "
             << fuelType() << "\n";
    }
    virtual ~Vehicle() {}
};

class ElectricCar : public Vehicle {
public:
    void start() override {
        cout << "Electric: silent start\n";
    }
    string fuelType() override {
        return "Electricity";
    }
};

class PetrolCar : public Vehicle {
public:
    void start() override {
        cout << "Petrol: Vroom!\n";
    }
    string fuelType() override {
        return "Petrol";
    }
};

int main() {
    ElectricCar ev;
    PetrolCar pc;
    ev.start(); ev.description();
    pc.start(); pc.description();
}
OUTPUT
Electric car: silent start ⚡
Vehicle runs on: Electricity
Petrol car: Vroom! 🔥
Vehicle runs on: Petrol
